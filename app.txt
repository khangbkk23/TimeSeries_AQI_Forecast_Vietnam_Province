import os
from datetime import datetime, timedelta
from flask import Flask, render_template, send_from_directory, flash, redirect, url_for
from apscheduler.schedulers.background import BackgroundScheduler

# Import the pipeline scripts
try: 
    from crawl import crawl_data
    from data_processor import drop_col, feature_engineering_and_preprocessing, prepare_data
    from plot_prediction import predict_next_7_days, visualize
except ValueError as e:
    print(f"Error while importing processing scripts: {e}. Please ensure that those file: crawl.py, data_processor.py, plot_prediction.py")

# Getting directories which needed for our demo web
data_dir = './data/'
json_dir = './JSON/'
ckpts_dir = './model/XGBoost/ckpts/'
plots_dir = './static/plots/'

# --- Config Flask ---
app = Flask(__name__)
app.secret_key = 'supersecretkey' # Required for flash messages

# Global variable to track the last run time
LAST_RUN_TIME = "Have not run yet!"
scheduler = None

def run_pipeline():
    """Executes the full data processing and plotting pipeline."""
    global LAST_RUN_TIME
    
    print("\n--- STARTING AUTOMATIC DATA PIPELINE ---")
    
    try:
        # 1. Cào dữ liệu
        crawl_data(data_dir)
        
        # 2. Xử lý và Dự đoán
        drop_col(data_dir)
        feature_engineering_and_preprocessing(data_dir)
        prepare_data(data_dir, data_dir + '/real_time_v1.pt')
        
        # 3. Vẽ biểu đồ
        dict_of_list = predict_next_7_days(ckpts_dir, data_dir)
        visualize(plots_dir, data_dir, dict_of_list)
        
        LAST_RUN_TIME = datetime.now().strftime("%H:%M:%S ngày %d/%m/%Y")
        print(f"--- PIPELINE COMPLETE. LAST RAN: {LAST_RUN_TIME} ---")
        
    except ImportError as e:
        error_msg = f"Error while running Pipeline: {e}"
        print(error_msg)

def start_scheduler():
    '''
        Initializes and starts the background schedulers
    '''
    global scheduler
    scheduler = BackgroundScheduler()

    # 1. Scheduler: Run every day at 05:00 AM
    scheduler.add_job(func=run_pipeline,
                      trigger='cron',
                      hour=5,
                      minute=0,
                      id='daily_crawl',
                      name='Daily Data Pipeline')
    
    # 2. Intial Run (For immediate demo purposes)
    # This job runs once, 5 seconds after startup, to generate initial plots.
    scheduler.add_job(func=run_pipeline,
                      trigger='date',
                      run_date=datetime.now() + timedelta(seconds=5),
                      id='initial_run',
                      name='Intial Demo Run')
    
    try:
        scheduler.start()
        print("APScheduler has been started")
    except Exception as e:
        print(f"Error while starting APScheduler: {e}")


# --- Flask Routes ---
@app.route('/')
def index():
    """
        Main route to display the prediction plots
    """
    plots_folder = 'plots'
    # List of expected plot files (Đã đổi sang .png)
    pollutant_names = ['CO', 'PM-10', 'PM-2-5', 'SO2']
    plot_files = [
        'aqi_CO.png',
        'aqi_PM-10.png',
        'aqi_PM-2-5.png',
        'aqi_SO2.png'
    ]
    
    # Check if files exist to avoid broken image links
    rendered_plots = []
    for i, f in enumerate(plot_files):
        path = os.path.join(app.static_folder, plots_folder, f)
        exists = os.path.exists(path)
        
        rendered_plots.append({
            'filename': f,
            'title': pollutant_names[i],
            'exists': exists
        })

    return render_template('index.html', 
                           plots=rendered_plots, 
                           last_run=LAST_RUN_TIME)

@app.route('/run_now')
def run_now():
    '''
        Manually trigger the pipeline for immediate update (for testing)
    '''
    global scheduler

    flash('Pipeline of data processing is running manually on background. Please wait 5-10 seconds and refresh page.', 'info')
    
    # Schedule the job to run in 1 second
    scheduler.add_job(func=run_pipeline,
                      trigger='date',
                      run_date=datetime.now() + timedelta(seconds=1),
                      name='Manual Run')
    return redirect(url_for('index'))

@app.route('/plots/<filename>')
def serve_plot(filename):
    """Serves the generated plot files (PDF or PNG)."""
    # Flask will automatically handle the correct MIME type for the PDF file
    return send_from_directory(os.path.join(app.static_folder, 'plots'), filename)

if __name__ == '__main__':
    start_scheduler()
    app.run(debug=True, host='127.0.0.1', port=5000)
